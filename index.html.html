<!-- Firebase SDK Modules -->
<script type="module">
import { initializeApp } from "https://www.gstatic.com/firebasejs/10.8.0/firebase-app.js";
import { getFirestore, collection, addDoc, onSnapshot, query, orderBy } from "https://www.gstatic.com/firebasejs/10.8.0/firebase-firestore.js";

// 🔴 Firebase Configuration
const firebaseConfig = {
apiKey: "YOUR_API_KEY",
authDomain: "photobs-app.firebaseapp.com",
projectId: "photobs-app",
storageBucket: "photobs-app.appspot.com",
messagingSenderId: "1234567890",
appId: "1:1234567890:web:abcdef123456"
};

// Firebase Initialize
const app = initializeApp(firebaseConfig);
const db = getFirestore(app);
const photosCollection = collection(db, "photos");

const gallery = document.getElementById('gallery');
const loadingMsg = document.getElementById('loading-msg');

// --- Real-time Live Synchronization with Firebase ---
const q = query(photosCollection, orderBy("timestamp", "desc"));

onSnapshot(q, (snapshot) => {
gallery.innerHTML = "";

if (snapshot.empty) {
gallery.innerHTML = "<div id='loading-msg'>No photos uploaded yet. Be the first to upload!</div>";
return;
}

snapshot.forEach((doc) => {
const data = doc.data();
const card = document.createElement('div');
card.className = 'photo-card';
card.innerHTML = `
<img src="${data.imageUrl}" alt="${data.title}">
<div class="photo-info">
<div class="photo-title">${data.title}</div>
<div class="photo-desc">${data.desc}</div>
<div class="card-footer">
<a href="${data.imageUrl}" download="${data.title}.jpg" target="_blank" class="download-btn">Download</a>
</div>
</div>
`;
gallery.appendChild(card);
});
}, (error) => {
console.log("Firebase not connected yet or using fallback demo mode.");
loadingMsg.innerText = "Showing sample gallery (Connect your Firebase API key for live database).";
});

// --- Upload Photo to Firebase ---
const submitPhotoBtn = document.getElementById('submit-photo-btn');
submitPhotoBtn.addEventListener('click', () => {
const fileInput = document.getElementById('photo-file');
const titleInput = document.getElementById('photo-title-input').value || "Untitled Photo";
const descInput = document.getElementById('photo-desc-input').value || "No description provided.";

if (fileInput.files && fileInput.files[0]) {
submitPhotoBtn.innerText = "Uploading...";
submitPhotoBtn.disabled = true;

const reader = new FileReader();
reader.onload = async function(e) {
const imageBase64 = e.target.result;

try {
await addDoc(photosCollection, {
title: titleInput,
desc: descInput,
imageUrl: imageBase64,
timestamp: new Date()
});

fileInput.value = "";
document.getElementById('photo-title-input').value = "";
document.getElementById('photo-desc-input').value = "";
document.getElementById('upload-modal').style.display = "none";
} catch (err) {
alert("Error uploading to Firebase. Make sure your Firebase Config is active!");
} finally {
submitPhotoBtn.innerText = "Add Photo";
submitPhotoBtn.disabled = false;
}
};
reader.readAsDataURL(fileInput.files[0]);
} else {
alert("Please select a photo first!");
}
});
</script>

<script>
// --- 1. Typewriter Animation ---
const textToType = "Welcome to Photobs";
let typeIndex = 0;
const typewriterEl = document.getElementById('typewriter-text');
const authorEl = document.getElementById('author-text');

function typeWriter() {
if (typeIndex < textToType.length) {
typewriterEl.innerHTML += textToType.charAt(typeIndex);
typeIndex++;
setTimeout(typeWriter, 100);
} else {
authorEl.style.opacity = "1";
setTimeout(() => {
const splash = document.getElementById('splash-screen');
splash.style.transition = "opacity 0.8s ease";
splash.style.opacity = "0";
setTimeout(() => splash.style.display = "none", 800);
}, 3800);
}
}

// --- 2. Fireworks Animation ---
const canvas = document.getElementById('fireworks');
const ctx = canvas.getContext('2d');
canvas.width = window.innerWidth;
canvas.height = window.innerHeight;

class Particle {
constructor(x, y, color) {
this.x = x;
this.y = y;
this.color = color;
this.velocity = {
x: (Math.random() - 0.5) * 6,
y: (Math.random() - 0.5) * 6
};
this.alpha = 1;
this.friction = 0.98;
}
draw() {
ctx.save();
ctx.globalAlpha = this.alpha;
ctx.beginPath();
ctx.arc(this.x, this.y, 3, 0, Math.PI * 2);
ctx.fillStyle = this.color;
ctx.fill();
ctx.restore();
}
update() {
this.velocity.x *= this.friction;
this.velocity.y *= this.friction;
this.x += this.velocity.x;
this.y += this.velocity.y;
this.alpha -= 0.015;
}
}

let particles = [];
const colors = ['#ff0055', '#00ffff', '#ffff00', '#00ff66', '#ff9900'];

function createFirework() {
const x = Math.random() * canvas.width;
const y = Math.random() * (canvas.height * 0.6);
const color = colors[Math.floor(Math.random() * colors.length)];
for (let i = 0; i < 30; i++) {
particles.push(new Particle(x, y, color));
}
}

setInterval(createFirework, 400);

function animateFireworks() {
ctx.fillStyle = 'rgba(8, 8, 16, 0.2)';
ctx.fillRect(0, 0, canvas.width, canvas.height);

particles.forEach((particle, index) => {
if (particle.alpha > 0) {
particle.update();
particle.draw();
} else {
particles.splice(index, 1);
}
});
requestAnimationFrame(animateFireworks);
}

window.addEventListener('DOMContentLoaded', () => {
typeWriter();
animateFireworks();
});

cards.forEach(card => {
const title = card.querySelector('.photo-title').innerText.toLowerCase();
const desc = card.querySelector('.photo-desc').innerText.toLowerCase();

card.style.display = "flex";

card.style.display = "none";
}
});
});
